use cosmwasm_std::StdError;
use thiserror::Error;

#[derive(Error, Debug)]
pub enum ContractError {
    #[error("{0}")]
    Std(#[from] StdError),

    #[error("unauthorized: only the configured relayer can update the price")]
    Unauthorized {},

    #[error("invalid price: {reason}")]
    InvalidPrice { reason: String },

    #[error("no price data yet")]
    NoPriceData {},
}
